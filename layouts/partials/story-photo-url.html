{{- /* URL of a chapter photo's screen-quality copy: //. photoBase is the local copy folder in development and Cloudflare R2 in production. Chapter resources and explicit data/photos URLs take priority. Drive URLs use the requested size (w1600 by default). Usage: partial "story-photo-url.html" (dict "page" .Page "name" "IMG_7284.jpg") */ -}} {{- $url := .name -}} {{- $resource := .page.Resources.GetMatch .name -}} {{- $trip := .page.Parent.File.ContentBaseName -}} {{- $chapter := .page.File.ContentBaseName -}} {{- $info := index (index site.Data.photos $trip | default dict) (partial "story-photo-key.html" (dict "page" .page "name" .name)) | default dict -}} {{- if $resource -}} {{- $url = $resource.RelPermalink -}} {{- else if $info.url -}} {{- $url = partial "gimg.html" (dict "url" $info.url "size" (.size | default "w1600")) -}} {{- else if not (hasPrefix .name "http") -}} {{- $url = printf "%s%s/%s" (site.Params.photoBase | default "r2/") .page.Parent.File.ContentBaseName (partial "story-photo-key.html" (dict "page" .page "name" .name)) -}} {{- if not (hasPrefix $url "http") -}}{{- $url = $url | absURL -}}{{- end -}} {{- end -}} {{- return $url -}}
